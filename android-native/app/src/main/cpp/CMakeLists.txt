cmake_minimum_required(VERSION 3.22.1)
project(aksara_sdturbo LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_OPENMP OFF CACHE BOOL "Avoid Android OpenMP dependency" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Portable ARM CPU build" FORCE)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_VULKAN OFF CACHE BOOL "" FORCE)
set(SD_OPENCL OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/third_party/stable-diffusion.cpp")
if(NOT EXISTS "${UPSTREAM}/include/stable-diffusion.h")
    message(FATAL_ERROR "Pinned stable-diffusion.cpp source missing; run native source checkout before the JNI build")
endif()
add_subdirectory("${UPSTREAM}" "${CMAKE_BINARY_DIR}/stable-diffusion-build")
add_library(aksara_sdturbo SHARED aksara_sdturbo.cpp)
target_include_directories(aksara_sdturbo PRIVATE "${UPSTREAM}/include")
target_link_libraries(aksara_sdturbo PRIVATE stable-diffusion log)

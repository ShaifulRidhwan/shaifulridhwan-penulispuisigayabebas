cmake_minimum_required(VERSION 3.22.1)
project(aksara_neural LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Keep the owner debug APK's CPU inference optimized while retaining native symbols.
set(CMAKE_C_FLAGS_DEBUG "-O3 -g")
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g")
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp")
if(NOT EXISTS "${UPSTREAM}/include/llama.h")
  message(FATAL_ERROR "Pinned llama.cpp source missing: run native source checkout")
endif()
add_subdirectory("${UPSTREAM}" "${CMAKE_BINARY_DIR}/llama-build")
add_library(aksara_neural SHARED aksara_neural.cpp)
target_include_directories(aksara_neural PRIVATE "${UPSTREAM}/include")
target_link_libraries(aksara_neural PRIVATE llama)
if(ANDROID)
  target_link_libraries(aksara_neural PRIVATE android log)
  target_link_options(aksara_neural PRIVATE "-Wl,-z,max-page-size=16384")
else()
  # Host JNI smoke testing uses the exact bridge; Android packaging is unchanged.
  find_package(JNI REQUIRED)
  target_include_directories(aksara_neural PRIVATE ${JNI_INCLUDE_DIRS})
endif()
